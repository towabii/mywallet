// スプレッドシートのシート名
const SHEET_NAME = 'Sheet1';

// Google スプレッドシートの 1 セルは最大 50,000 文字です。
// 余裕を持たせ、JSON は 45,000 文字ずつ C 列以降に分割して保存します。
const JSON_CHUNK_SIZE = 45000;
const STORAGE_VERSION = 'chunked-v1';
const METADATA_COLUMN = 2; // B列
const FIRST_CHUNK_COLUMN = 3; // C列

// デプロイ確認用: /exec をブラウザで開いたときにJSONが返れば、URLと公開設定は有効です。
function doGet() {
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}

// フロントエンドからのPOSTリクエストを処理するメイン関数
function doPost(e) {
  let response;
  try {
    const request = JSON.parse(e.postData.contents);
    switch (request.action) {
      case 'signup': response = signup(request.payload); break;
      case 'login': response = login(request.payload); break;
      case 'getUserData': response = getUserData(request.payload); break;
      case 'updateUserData': response = updateUserData(request.payload.userId, request.payload.userData, request.payload.authToken); break;
      case 'syncData': response = syncData(request.payload); break;
      case 'changeEmail': response = changeEmail(request.payload); break;
      case 'changePassword': response = changePassword(request.payload); break;
      case 'getBackupKey': response = getBackupKey(request.payload); break;
      default: response = { status: 'error', message: '無効なアクションです。' };
    }
  } catch (error) {
    response = { status: 'error', message: 'サーバーエラー: ' + error.message };
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheet_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error('シート「' + SHEET_NAME + '」が見つかりません。');
  return sheet;
}

// 読み書きとも同じロックを使います。保存途中の断片を別リクエストが読むことを防ぎます。
function withScriptLock_(callback) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    return callback();
  } finally {
    lock.releaseLock();
  }
}

const SESSION_TTL_SECONDS = 21600;
function createSession_(userId) {
  const token = Utilities.getUuid() + Utilities.getUuid();
  CacheService.getScriptCache().put('session:' + token, String(userId), SESSION_TTL_SECONDS);
  return token;
}

function assertSession_(userId, token) {
  const sessionUserId = token && CacheService.getScriptCache().get('session:' + token);
  if (!sessionUserId || String(sessionUserId) !== String(userId)) throw new Error('ログインの有効期限が切れました。もう一度ログインしてください。');
}

function isChunkedMetadata_(value) {
  return value && value.storageVersion === STORAGE_VERSION;
}

function splitIntoChunks_(text) {
  const chunks = [];
  for (let start = 0; start < text.length; start += JSON_CHUNK_SIZE) {
    chunks.push(text.slice(start, start + JSON_CHUNK_SIZE));
  }
  // JSON.stringify は空文字列にならないものの、保存形式としては空でも1セル確保します。
  return chunks.length ? chunks : [''];
}

function sha256_(text) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    text,
    Utilities.Charset.UTF_8
  );
  return bytes.map(function(byte) {
    const value = byte < 0 ? byte + 256 : byte;
    return ('0' + value.toString(16)).slice(-2);
  }).join('');
}

function ensureColumnCount_(sheet, requiredLastColumn) {
  const currentColumnCount = sheet.getMaxColumns();
  if (currentColumnCount < requiredLastColumn) {
    sheet.insertColumnsAfter(currentColumnCount, requiredLastColumn - currentColumnCount);
  }
}

// 行のユーザーデータを復元します。断片数・長さ・ハッシュ・IDを検証するため、
// 別行のデータや古い断片を誤って組み合わせて返すことはありません。
function readUserAtRow_(sheet, row, metadata) {
  if (!isChunkedMetadata_(metadata)) return metadata; // 旧形式（B列にJSON全体）の読み込み互換性

  const chunkCount = Number(metadata.chunkCount);
  if (!Number.isInteger(chunkCount) || chunkCount < 1) {
    throw new Error('ユーザーデータの分割情報が不正です。');
  }

  const chunks = sheet.getRange(row, FIRST_CHUNK_COLUMN, 1, chunkCount).getValues()[0];
  if (chunks.some(function(chunk) { return typeof chunk !== 'string'; })) {
    throw new Error('ユーザーデータの一部が見つかりません。');
  }

  const json = chunks.join('');
  if (json.length !== Number(metadata.dataLength) || sha256_(json) !== metadata.checksum) {
    throw new Error('ユーザーデータの整合性を確認できません。');
  }

  const user = JSON.parse(json);
  if (String(user.id) !== String(metadata.id) || user.email !== metadata.email) {
    throw new Error('ユーザーデータの所有者が一致しません。');
  }
  return user;
}

// B列だけを先に走査し、対象行が見つかった時だけC列以降のJSONを復元します。
function findUserBy_(sheet, key, value) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;

  const metadataCells = sheet.getRange(2, METADATA_COLUMN, lastRow - 1, 1).getValues();
  for (let index = 0; index < metadataCells.length; index++) {
    let metadata;
    try {
      metadata = JSON.parse(metadataCells[index][0]);
    } catch (e) {
      continue;
    }

    const fieldValue = metadata[key];
    if (String(fieldValue) !== String(value)) continue;

    const row = index + 2;
    const user = readUserAtRow_(sheet, row, metadata);
    // chunked-v1 のB列は検索用コピーなので、復元後のJSONも必ず照合します。
    if (String(user[key]) !== String(value)) {
      throw new Error('ユーザー検索情報の整合性を確認できません。');
    }
    return { user: user, row: row, metadata: metadata };
  }
  return null;
}

// 1ユーザーのJSON断片は必ず同じ行のC列以降にだけ保存します。
function writeUserAtRow_(sheet, row, user, previousMetadata) {
  if (!user || typeof user !== 'object' || Array.isArray(user)) {
    throw new Error('保存するユーザーデータが不正です。');
  }
  const json = JSON.stringify(user);
  const chunks = splitIntoChunks_(json);
  const metadata = {
    storageVersion: STORAGE_VERSION,
    id: user.id,
    email: user.email,
    passwordHash: user.passwordHash,
    backupKeyHash: user.backupKeyHash,
    updatedAt: user.updatedAt,
    chunkCount: chunks.length,
    dataLength: json.length,
    checksum: sha256_(json)
  };

  // 先に断片を書き、最後にB列のメタデータを切り替えます。
  // ロック中なので、他のAPI呼び出しが中間状態を観測することはありません。
  ensureColumnCount_(sheet, FIRST_CHUNK_COLUMN + chunks.length - 1);
  sheet.getRange(row, FIRST_CHUNK_COLUMN, 1, chunks.length).setValues([chunks]);
  sheet.getRange(row, METADATA_COLUMN).setValue(JSON.stringify(metadata));

  const oldCount = isChunkedMetadata_(previousMetadata) ? Number(previousMetadata.chunkCount) : 0;
  if (Number.isInteger(oldCount) && oldCount > chunks.length) {
    sheet.getRange(row, FIRST_CHUNK_COLUMN + chunks.length, 1, oldCount - chunks.length).clearContent();
  }
}

function saveExistingUser_(sheet, found, user) {
  // リクエスト内のuser.idを書換先の行IDに固定し、別ユーザーのIDで上書きされないようにします。
  user.id = found.user.id;
  writeUserAtRow_(sheet, found.row, user, found.metadata);
}

// 新規登録
function signup(payload) {
  if (!payload.email || !payload.passwordHash || !payload.backupKeyHash) {
    return { status: 'error', message: '情報が不足しています。' };
  }

  return withScriptLock_(function() {
    const sheet = getSheet_();
    if (findUserBy_(sheet, 'email', payload.email)) {
      return { status: 'error', message: 'このメールアドレスは既に使用されています。' };
    }

    let userId = new Date().getTime();
    while (findUserBy_(sheet, 'id', userId)) userId++;
    const newUser = {
      id: userId,
      email: payload.email,
      passwordHash: payload.passwordHash,
      backupKeyHash: payload.backupKeyHash,
      updatedAt: Date.now(),
      data: { accounts: [], categories: [], transactions: [] }
    };
    sheet.appendRow([userId]);
    writeUserAtRow_(sheet, sheet.getLastRow(), newUser, null);
    return { status: 'success', message: 'アカウントが作成されました。' };
  });
}

// ログイン
function login(payload) {
  return withScriptLock_(function() {
    const found = findUserBy_(getSheet_(), 'email', payload.email);
    if (found && found.user.passwordHash === payload.passwordHash) {
      return { status: 'success', userId: found.user.id, userData: found.user, authToken: createSession_(found.user.id) };
    }
    return { status: 'error', message: 'メールアドレスまたはパスワードが間違っています。' };
  });
}

// ユーザーデータ取得
function getUserData(payload) {
  return withScriptLock_(function() {
    assertSession_(payload.userId, payload.authToken);
    const found = findUserBy_(getSheet_(), 'id', payload.userId);
    return found
      ? { status: 'success', userData: found.user }
      : { status: 'error', message: 'ユーザーが見つかりません。' };
  });
}

// 同期機能（タイムスタンプによる競合チェックあり）
function syncData(payload) {
  return withScriptLock_(function() {
    assertSession_(payload.userId, payload.authToken);
    const sheet = getSheet_();
    const found = findUserBy_(sheet, 'id', payload.userId);
    if (!found) return { status: 'error', message: 'ユーザーが見つかりません。' };

    if (found.user.updatedAt && payload.lastSyncedAt && found.user.updatedAt > payload.lastSyncedAt) {
      return { status: 'conflict', serverData: found.user };
    }

    payload.userData.updatedAt = Date.now();
    saveExistingUser_(sheet, found, payload.userData);
    return { status: 'success', userData: payload.userData };
  });
}

// ユーザーデータ強制更新
function updateUserData(userId, userData) {
  const authToken = arguments[2];
  return withScriptLock_(function() {
    assertSession_(userId, authToken);
    const sheet = getSheet_();
    const found = findUserBy_(sheet, 'id', userId);
    if (!found) return { status: 'error', message: 'ユーザーが見つかりません。' };

    saveExistingUser_(sheet, found, userData);
    return { status: 'success', message: 'データが更新されました。' };
  });
}

// メールアドレス変更
function changeEmail(payload) {
  return withScriptLock_(function() {
    assertSession_(payload.userId, payload.authToken);
    const sheet = getSheet_();
    const found = findUserBy_(sheet, 'id', payload.userId);
    if (!found || found.user.passwordHash !== payload.passwordHash) {
      return { status: 'error', message: 'パスワードが間違っています。' };
    }
    const duplicate = findUserBy_(sheet, 'email', payload.newEmail);
    if (duplicate && duplicate.row !== found.row) {
      return { status: 'error', message: 'このメールアドレスは既に使用されています。' };
    }

    found.user.email = payload.newEmail;
    found.user.updatedAt = Date.now();
    saveExistingUser_(sheet, found, found.user);
    return { status: 'success', message: 'データが更新されました。' };
  });
}

// パスワード変更
function changePassword(payload) {
  return withScriptLock_(function() {
    assertSession_(payload.userId, payload.authToken);
    const sheet = getSheet_();
    const found = findUserBy_(sheet, 'id', payload.userId);
    if (!found || found.user.passwordHash !== payload.currentPasswordHash) {
      return { status: 'error', message: '現在のパスワードが間違っています。' };
    }
    if (found.user.backupKeyHash !== payload.backupKeyHash) {
      return { status: 'error', message: 'バックアップキーが間違っています。' };
    }

    found.user.passwordHash = payload.newPasswordHash;
    found.user.updatedAt = Date.now();
    saveExistingUser_(sheet, found, found.user);
    return { status: 'success', message: 'データが更新されました。' };
  });
}

// バックアップキー取得
function getBackupKey(payload) {
  return withScriptLock_(function() {
    assertSession_(payload.userId, payload.authToken);
    const found = findUserBy_(getSheet_(), 'id', payload.userId);
    if (found && found.user.passwordHash === payload.passwordHash) {
      return { status: 'success', backupKeyHash: found.user.backupKeyHash };
    }
    return { status: 'error', message: 'パスワードが間違っています。' };
  });
}
